WITH current_ticket AS (
  SELECT COALESCE(qc_logs, '[]'::jsonb) AS logs
  FROM public.tickets
  WHERE ticked_id = '{{Qc_Table.triggeredRow.ticked_id}}'
)
UPDATE public.tickets t
SET qc_logs = c.logs || jsonb_build_array(
  jsonb_build_object(
    'qc_start_time', to_char(CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Kolkata', 'YYYY-MM-DD HH24:MI:SS'),
    'qc_end_time', NULL,
    'qc_passed', NULL,
    'qc_tag', NULL,
    'qc_description', NULL,
    'qc_notes', NULL,
    'time_difference', NULL
  )
)
FROM current_ticket c
WHERE t.ticked_id = '{{Qc_Table.triggeredRow.ticked_id}}'
  AND (
    jsonb_array_length(c.logs) = 0
    OR (c.logs -> (jsonb_array_length(c.logs) - 1) ->> 'qc_end_time') IS NOT NULL
  )
RETURNING t.ticked_id;
