SELECT work_start_time 
FROM public.tickets
WHERE 
  (
    (email = '{{ Tickets.triggeredRow.email }}' AND ticked_id = '{{ Tickets.triggeredRow.ticked_id }}')
    OR
    (email = '{{ TicketsonProgress.triggeredRow.email }}' AND ticked_id = '{{ TicketsonProgress.triggeredRow.ticked_id }}')
    OR
    (email = '{{ TicketsCompletedInsights.triggeredRow.email }}' AND ticked_id = '{{ TicketsCompletedInsights.triggeredRow.ticked_id }}')
    OR
    (email = '{{ TicketsCompletedDashboard.triggeredRow.email }}' AND ticked_id = '{{ TicketsCompletedDashboard.triggeredRow.ticked_id }}')
    OR
    (email = '{{ TicketsCompletedAutomation.triggeredRow.email }}' AND ticked_id = '{{ TicketsCompletedAutomation.triggeredRow.ticked_id }}')
    OR
    (email = '{{ TicketsCompletedAI.triggeredRow.email }}' AND ticked_id = '{{ TicketsCompletedAI.triggeredRow.ticked_id }}')
  );
